cmake_minimum_required(VERSION 3.20)
project(dali-buxby VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUXBY_HOST_PREVIEW "Build the Ubuntu visual preview instead of Team member libraries" OFF)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UI REQUIRED IMPORTED_TARGET dali2-ui-foundation dali2-adaptor dali2-core)
pkg_check_modules(CHART REQUIRED IMPORTED_TARGET dali2-ui-components)
find_program(MSGFMT msgfmt REQUIRED)

if(BUXBY_HOST_PREVIEW)
  set(BUXBY_LOCALE_DIR "${CMAKE_CURRENT_BINARY_DIR}/locale")
else()
  set(BUXBY_LOCALE_DIR "${CMAKE_INSTALL_PREFIX}/res/locale")
endif()
add_library(buxby-style STATIC src/style.cpp src/catalog.cpp src/window-navigation.cpp src/scene-motion.cpp)
target_link_libraries(buxby-style PUBLIC PkgConfig::UI)
target_compile_definitions(buxby-style PUBLIC HIDE_DALI_INTERNALS PRIVATE BUXBY_LOCALE_DIR="${BUXBY_LOCALE_DIR}")
target_include_directories(buxby-style PUBLIC src)
target_compile_options(buxby-style PRIVATE -Wall -Wextra -Wpedantic)
foreach(member main day activity)
  add_library(buxby-${member}-scene STATIC src/${member}-window.cpp)
  target_link_libraries(buxby-${member}-scene PUBLIC buxby-style)
  target_compile_options(buxby-${member}-scene PRIVATE -Wall -Wextra -Wpedantic)
  if(member STREQUAL "activity")
    target_link_libraries(buxby-${member}-scene PUBLIC PkgConfig::CHART)
  endif()
  foreach(locale en_US ko_KR ar_AE)
    set(mo "${CMAKE_CURRENT_BINARY_DIR}/locale/${locale}/LC_MESSAGES/dali-buxby-${member}.mo")
    add_custom_command(OUTPUT "${mo}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/locale/${locale}/LC_MESSAGES"
      COMMAND ${MSGFMT} --check --check-format -o "${mo}" "${CMAKE_CURRENT_SOURCE_DIR}/po/dali-buxby-${member}/${locale}.po"
      DEPENDS "po/dali-buxby-${member}/${locale}.po" VERBATIM)
    list(APPEND catalogs "${mo}")
    install(FILES "${mo}" DESTINATION "res/locale/${locale}/LC_MESSAGES")
  endforeach()
endforeach()
add_custom_target(buxby-catalogs ALL DEPENDS ${catalogs})

if(BUXBY_HOST_PREVIEW)
  find_package(X11 REQUIRED)
  add_executable(buxby-preview tools/preview/main.cpp)
  target_link_libraries(buxby-preview PRIVATE buxby-main-scene buxby-day-scene buxby-activity-scene X11::X11)
  add_dependencies(buxby-preview buxby-catalogs)
else()
  # Target entry points and the installed platform API check are configured below.
  include(cmake/team-targets.cmake)
endif()

if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME reference-contract COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/check-reference.py"
    "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
endif()
